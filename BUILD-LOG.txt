# toolchain-export ワークフローの設定パネル（エージェント側から書き換えて push するだけ）。
#
# 背景: 開発サンドボックスに Rust ツールチェーンが無く、static.rust-lang.org にも
# crates.io にも到達できない（PLANS.md §9.3）。そこで Actions 側でツールチェーンと
# 依存クレートのソースをまとめ、孤立ブランチに commit して git 経由で持ち出す。
#
# 不要になったら ENABLED=0（またはこのファイルと workflow を削除）にして、
#   git push origin --delete cache/rust-toolchain
# で生成物ブランチも消す。main にも作業ブランチにも影響しない。

# 0 にすると checkout 直後に終了する（Actions 使用時間の節約）。
ENABLED=1

# 生成物を push する孤立ブランチ（orphan）。履歴は作業ブランチと完全に分離される。
CACHE_BRANCH=cache/rust-toolchain

# GitHub は 1 ファイル 100MB までしか push できない。この大きさで分割する。
# 45MB にしてあるのは、大きな pack を 1 リクエストで送ると
# "RPC failed; curl 55 Send failure: Broken pipe" で切れたため（1 チャンクずつ push する）。
CHUNK_MB=45

# 入れるツールチェーン。CI（dtolnay/rust-toolchain@stable）と同じ構成にする。
TOOLCHAIN=stable
PROFILE=minimal
COMPONENTS=rustfmt,clippy

# 追加ターゲット（CI の wasm ジョブと同じ）。空にすると host だけになる。
TARGETS=wasm32-unknown-unknown

# 依存クレートのソースを cargo vendor で同梱する（サンドボックスは crates.io 到達不可）。
VENDOR=1

# 同梱物だけで実際にビルド＆テストして、Debian 12 互換を push 前に確認する。
# Actions 使用時間を削りたいときは SMOKE=0、または SMOKE_CMD を軽いコマンドにする。
SMOKE=1
# eval されるので && やパイプも書ける。CI の test ジョブと wasm ジョブに相当する。
SMOKE_CMD="cargo test --locked --all-targets && cargo check --locked --target wasm32-unknown-unknown"

# cache_key は (このファイル + rust-toolchain.toml + Cargo.lock + SALT) のハッシュ。
# 同じ key の生成物が既にブランチにあれば何もしない。中身を変えずに作り直したいときや
# stable のバージョンが上がったあとに更新したいときは SALT を増やす。
SALT=1

# 1 にすると cache_key が一致しても強制的に作り直す。
FORCE=0
